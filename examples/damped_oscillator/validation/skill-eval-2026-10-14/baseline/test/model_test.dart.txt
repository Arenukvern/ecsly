import 'dart:math' as math;
import 'package:flutter_test/flutter_test.dart';
import 'package:damped_lab/model.dart';

void main() {
  test(
    'starts paused at initial conditions with complete four-second plot',
    () {
      final m = Oscillator();
      expect(m.current.time, 0);
      expect(m.current.x, 1);
      expect(m.current.v, 0);
      expect(m.damping, .3);
      expect(m.samples.length, 241);
      expect(m.samples.last.time, 4);
      expect(m.current.energy, closeTo(omega * omega / 2, 1e-12));
    },
  );
  test('step advances one fixed interval and matches preview', () {
    final m = Oscillator();
    m.step();
    expect(m.current.time, fixedStep);
    expect(m.current.x, m.samples[1].x);
    expect(m.current.v, m.samples[1].v);
  });
  test('undamped evolution agrees with analytic solution', () {
    final m = Oscillator()..reset(0);
    for (var i = 0; i < 240; i++) {
      m.step();
    }
    expect(m.current.x, closeTo(math.cos(omega * 4), 1e-4));
    expect(m.current.v, closeTo(-omega * math.sin(omega * 4), 2e-4));
    expect(m.current.energy, closeTo(omega * omega / 2, .001));
  });
  test(
    'damped energy decreases and overdamped trajectory does not cross zero',
    () {
      for (final damping in [.3, 1.2]) {
        final m = Oscillator()..reset(damping);
        for (var i = 1; i < m.samples.length; i++) {
          expect(
            m.samples[i].energy,
            lessThanOrEqualTo(m.samples[i - 1].energy + 1e-9),
          );
          if (damping > 1) {
            expect(m.samples[i].x, greaterThan(0));
          }
        }
      }
    },
  );
  test('presets reset state; reset retains damping; selection is linked', () {
    final m = Oscillator()
      ..step()
      ..nextSample()
      ..reset(1.2);
    expect(m.tick, 0);
    expect(m.selectedIndex, 0);
    m.step();
    m.reset();
    expect(m.damping, 1.2);
    m.nextSample();
    expect(m.snapshot()['plotSelection'], {'time': 1, 'phase': 1});
    expect(m.current.time, 0);
    for (var i = 0; i < 240; i++) {
      m.nextSample();
    }
    expect(m.selectedIndex, 0);
  });
}
