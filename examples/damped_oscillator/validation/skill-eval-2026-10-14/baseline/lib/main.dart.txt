class OscillatorPainter extends CustomPainter {
  OscillatorPainter(this.samples, this.current, this.selected, this.phase);
  final List<Observation> samples;
  final Observation current;
  final int selected;
  final bool phase;
  void label(Canvas c, String text, Offset at, {Color color = ink}) {
    final p = TextPainter(
      text: TextSpan(
        text: text,
        style: TextStyle(color: color, fontSize: 12),
      ),
      textDirection: TextDirection.ltr,
    )..layout();
    p.paint(c, at);
  }

  @override
  void paint(Canvas canvas, Size size) {
    final area = Rect.fromLTRB(44, 14, size.width - 16, size.height - 44);
    final xmin = phase ? -1.1 : 0.0;
    final xmax = phase ? 1.1 : 4.0;
    final ymin = phase ? -7.0 : -1.1;
    final ymax = phase ? 7.0 : 1.1;
    Offset map(double x, double y) => Offset(
      area.left + (x - xmin) / (xmax - xmin) * area.width,
      area.bottom - (y - ymin) / (ymax - ymin) * area.height,
    );
    Offset point(Observation o) => map(phase ? o.x : o.time, phase ? o.v : o.x);
    final grid = Paint()
      ..color = const Color(0xFF304653)
      ..strokeWidth = 1;
    for (var i = 0; i <= 4; i++) {
      final x = xmin + i * (xmax - xmin) / 4;
      final y = ymin + i * (ymax - ymin) / 4;
      canvas.drawLine(map(x, ymin), map(x, ymax), grid);
      canvas.drawLine(map(xmin, y), map(xmax, y), grid);
      label(
        canvas,
        x.toStringAsFixed(phase ? 1 : 0),
        Offset(map(x, ymin).dx - 7, area.bottom + 8),
      );
      label(canvas, y.toStringAsFixed(1), Offset(0, map(xmin, y).dy - 6));
    }
    final axes = Paint()
      ..color = const Color(0xFF7994A5)
      ..strokeWidth = 1.3;
    canvas.drawLine(map(xmin, 0), map(xmax, 0), axes);
    if (phase) {
      canvas.drawLine(map(0, ymin), map(0, ymax), axes);
    }
    canvas.save();
    canvas.clipRect(area);
    final path = Path()
      ..moveTo(point(samples.first).dx, point(samples.first).dy);
    for (final s in samples.skip(1)) {
      path.lineTo(point(s).dx, point(s).dy);
    }
    canvas.drawPath(
      path,
      Paint()
        ..color = teal
        ..strokeWidth = 2.5
        ..style = PaintingStyle.stroke,
    );
    canvas.drawCircle(
      point(samples[selected]),
      7,
      Paint()
        ..color = amber
        ..style = PaintingStyle.stroke
        ..strokeWidth = 3,
    );
    canvas.drawCircle(point(current), 3.5, Paint()..color = Colors.white);
    canvas.restore();
    label(
      canvas,
      phase ? 'Displacement (m)' : 'Time (s)',
      Offset(area.left + area.width / 2 - 45, size.height - 16),
    );
    label(canvas, phase ? 'v (m/s)' : 'x (m)', const Offset(2, 0), color: teal);
  }

  @override
  bool shouldRepaint(covariant OscillatorPainter oldDelegate) => true;
}
